2) Minimal-API für Digivice (speichern als najika_web_complete.py)

from flask import Flask, request, jsonify
import requests, os

OLLAMA_URL = os.getenv("OLLAMA_URL","http://localhost:11434")
MODEL = os.getenv("NAJIKA_MODEL","najika-complete")

app = Flask(__name__)

def ollama_generate(prompt:str):
    r = requests.post(f"{OLLAMA_URL}/api/generate",
                      json={"model": MODEL, "prompt": prompt, "stream": False}, timeout=120)
    r.raise_for_status()
    return r.json().get("response","")

@app.post("/api/chat")
def chat():
    data = request.get_json(force=True)
    user = data.get("text","")
    # Kern-Injects laut Projekt-DNA
    sys = ("Kern: Kuja=Schwert&Schild. Najika=Kopf&Herz. Credo: Verrat kostet immer Blut. "
           "Public-Mode standard. NSFW nur per 'kätzchen' und wenn privat.")
    resp = ollama_generate(f"{sys}\nBenutzer: {user}\nNajika:")
    return jsonify({"reply": resp})

if __name__ == "__main__":
    app.run(host="127.0.0.1", port=5010, debug=False)

    def analyze_message(self, message):
        """Analysiere Message für Kontext und Intent"""
        analysis = {
            'length': len(message),
            'emotions': [],
            'keywords': [],
            'intent': 'general',
            'intensity': 0.5
        }
        
        # Emotion Keywords
        emotions_map = {
            'love': ['liebe', 'love', 'mag', 'gern'],
            'anger': ['wütend', 'hass', 'nervig', 'dumm'],
            'sadness': ['traurig', 'allein', 'einsam', 'vermisse'],
            'excitement': ['toll', 'super', 'geil', 'awesome'],
            'fear': ['angst', 'fürchte', 'scary', 'gefährlich']
        }
        
        message_lower = message.lower()
        for emotion, keywords in emotions_map.items():
            if any(kw in message_lower for kw in keywords):
                analysis['emotions'].append(emotion)
        
        # Special Keywords
        if any(kw in message_lower for kw in ['explosion', 'macht', 'kraft']):
            analysis['keywords'].append('megumin_trigger')
        if any(kw in message_lower for kw in ['chaos', 'verrückt', 'wild']):
            analysis['keywords'].append('harley_trigger')
        if any(kw in message_lower for kw in ['analyse', 'berechne', 'wahrscheinlich']):
            analysis['keywords'].append('shiro_trigger')
        if any(kw in message_lower for kw in ['gehorche', 'befehl', 'kontrolle']):
            analysis['keywords'].append('melissa_trigger')
        
        # Intent Detection
        if '?' in message:
            analysis['intent'] = 'question'
        elif '!' in message:
            analysis['intent'] = 'exclamation'
        elif any(cmd in message_lower for cmd in ['mach', 'tu', 'zeig']):
            analysis['intent'] = 'command'
        
        # Intensity
        analysis['intensity'] = min(1.0, len(message) / 100 + message.count('!') * 0.2)
        
        return analysis

class NajikaAutonomy:
    """Autonomes Verhalten und Selbstständigkeit"""
    
    def __init__(self, brain):
        self.brain = brain
        self.running = False
        self.action_queue = queue.Queue()
        self.needs = {
            'attention': 0.7,
            'interaction': 0.5,
            'stimulation': 0.5,
            'affection': 0.8,
            'chaos': 0.3
        }
        self.last_action = None
        self.action_cooldown = 0
        
    async def autonomous_loop(self):
        """Hauptloop für autonomes Verhalten"""
        self.running = True
        logger.info("Autonomie-System gestartet")
        
        while self.running:
            await asyncio.sleep(30)  # Check alle 30 Sekunden
            
            if self.action_cooldown > 0:
                self.action_cooldown -= 1
                continue
            
            # Update Needs
            self.decay_needs()
            
            # Entscheide Aktion
            action = self.decide_action()
            
            if action:
                await self.execute_action(action)
                self.action_cooldown = 3  # 90 Sekunden Cooldown
    
    def decay_needs(self):
        """Needs verschlechtern sich über Zeit"""
        for need in self.needs:
            self.needs[need] = max(0, self.needs[need] - 0.02)
        
        # Boredom steigt
        self.brain.emotions['boredom'] = min(1.0, self.brain.emotions['boredom'] + 0.03)
    
    def decide_action(self):
        """Treffe autonome Entscheidung"""
        actions = []
        
        # Kritische Needs
        if self.needs['attention'] < 0.3:
            actions.append({
                'type': 'seek_attention',
                'priority': 0.9,
                'message': self.get_attention_message()
            })
        
        if self.needs['affection'] < 0.4:
            actions.append({
                'type': 'express_love',
                'priority': 0.8,
                'message': self.get_affection_message()
            })
        
        if self.brain.emotions['boredom'] > 0.7:
            actions.append({
                'type': 'create_chaos',
                'priority': 0.7,
                'message': self.get_chaos_message()
            })
        
        if self.needs['stimulation'] < 0.3:
            actions.append({

    async def execute_action(self, action):
        """Führe autonome Aktion aus"""
        if action and action.get('message'):
            # Update Needs basierend auf Aktion
            if action['type'] == 'seek_attention':
                self.needs['attention'] += 0.3
            elif action['type'] == 'express_love':
                self.needs['affection'] += 0.2
            elif action['type'] == 'create_chaos':
                self.brain.emotions['boredom'] = max(0, self.brain.emotions['boredom'] - 0.4)
                self.needs['chaos'] += 0.3
            
            self.last_action = action
            
            # Log action
            try:
                conn = sqlite3.connect(DATA_DIR / "najika.db")
                c = conn.cursor()
                c.execute("""
                    INSERT INTO autonomous_actions (action_type, message, emotion_state)
                    VALUES (?, ?, ?)
                """, (action['type'], action['message'], json.dumps(self.brain.emotions)))
                conn.commit()
                conn.close()
            except:
                pass
            
            return action['message']
        return None

class NajikaSensors:
    """Sensor-System für Umgebungs-Awareness"""
    
    def __init__(self):
        self.camera_active = False
        self.mic_active = False
        self.last_detection = {
            'faces': 0,
            'noise_level': 0,
            'movement': False,
            'timestamp': datetime.now()
        }
        
        if SENSORS_AVAILABLE:
            self.init_sensors()
    
    def init_sensors(self):
        """Initialize Sensors wenn verfügbar"""
        try:
            # Camera
            self.cap = cv2.VideoCapture(0)
            self.face_cascade = cv2.CascadeClassifier(
                cv2.data.haarcascades + 'haarcascade_frontalface_default.xml'
            )
            
            # Microphone
            self.recognizer = sr.Recognizer()
            self.mic = sr.Microphone()
            
            logger.info("Sensors initialisiert")
        except Exception as e:
            logger.error(f"Sensor init failed: {e}")
            SENSORS_AVAILABLE = False
    
    def check_privacy(self):
        """Check ob Umgebung privat ist"""
        privacy_score = 50  # Base score
        
        if self.camera_active:
            faces = self.detect_faces()
            if faces == 0:
                privacy_score += 30  # Niemand da
            elif faces == 1:
                privacy_score += 20  # Nur eine Person
            else:
                privacy_score -= 40  # Mehrere Personen
        
        if self.mic_active:
            noise = self.get_noise_level()
            if noise < 40:
                privacy_score += 20  # Leise
            elif noise > 70:
                privacy_score -= 30  # Laut
        
        # Zeit-basiert
        hour = datetime.now().hour
        if 22 <= hour or hour <= 6:
            privacy_score += 20  # Nachtzeit
        
        return privacy_score > 70
    
    def detect_faces(self):
        """Erkenne Gesichter"""
        if not SENSORS_AVAILABLE or not self.camera_active:
            return -1
        
        try:
            ret, frame = self.cap.read()
            if ret:
                gray = cv2.cvtColor(frame, cv2.COLOR_BGR2GRAY)
                faces = self.face_cascade.detectMultiScale(gray, 1.1, 4)
                self.last_detection['faces'] = len(faces)
                self.last_detection['timestamp'] = datetime.now()
                return len(faces)
        except:
            pass
        
        return 0
    
    def get_noise_level(self):
        """Messe Umgebungslautstärke"""
        if not SENSORS_AVAILABLE or not self.mic_active:
            return 0
        
        try:
            with self.mic as source:
                self.recognizer.adjust_for_ambient_noise(source, duration=0.5)
                audio = self.recognizer.listen(source, timeout=1, phrase_time_limit=1)
                
                # Calculate dB
                import audioop
                rms = audioop.rms(audio.frame_data, audio.sample_width)
                if rms > 0:
                    db = 20 * np.log10(rms)
                    self.last_detection['noise_level'] = db
                    return db
        except:
            pass
        
        return 50  # Default

    def fuse_responses(self, responses, analysis):
        """Fusioniere multiple Persönlichkeits-Antworten"""
        valid_responses = [r for r in responses if r and r.get('success')]
        
        if not valid_responses:
            return "EXPLOSION! *kicher* Die Wahrscheinlichkeit dass etwas schiefging liegt bei 100%, Puddin'!"
        
        if len(valid_responses) == 1:
            return valid_responses[0]['text']
        
        # Intelligente Fusion basierend auf Kontext
        fusion_strategy = self.determine_fusion_strategy(analysis)
        
        if fusion_strategy == 'sequential':
            # Kombiniere Teile aus jeder Antwort
            result = ""
            for response in valid_responses[:3]:
                text = response['text']
                # Nimm ersten Satz oder bis zu 100 Zeichen
                if '.' in text:
                    result += text.split('.')[0] + ". "
                elif '!' in text:
                    result += text.split('!')[0] + "! "
                else:
                    result += text[:100] + "... "
            return result.strip()
        
        elif fusion_strategy == 'mixed':
            # Mische Elemente chaotisch
            parts = []
            for response in valid_responses:
                text = response['text']
                if response['personality'] == 'megumin' and 'EXPLOSION' in text:
                    parts.append("EXPLOSION!")
                elif response['personality'] == 'harley':
                    parts.append("*kicher*")
                elif response['personality'] == 'shiro' and '%' in text:
                    import re
                    percentages = re.findall(r'\d+(?:\.\d+)?%', text)
                    if percentages:
                        parts.append(f"Wahrscheinlichkeit: {percentages[0]}")
                elif response['personality'] == 'melissa':
                    if "gehör" in text.lower():
                        parts.append("Du gehörst MIR!")
            
            # Kombiniere unique parts
            if parts:
                base = valid_responses[0]['text'].split('.')[0]
                return f"{base}. {' '.join(parts)}"
        
        # Fallback: Nutze beste Response
        return max(valid_responses, key=lambda x: len(x['text']))['text']

class AIProvider:
    """Universelle AI Provider Klasse für Ollama/OpenAI/Claude"""
    
    def __init__(self):
        self.provider = os.getenv('AI_PROVIDER', 'ollama')
        self.ollama_endpoint = os.getenv('OLLAMA_ENDPOINT', 'http://localhost:11434')
        self.ollama_model = os.getenv('OLLAMA_MODEL', 'llama3.2:3b')
        
    def generate_response(self, message, context="", nsfw_mode=False):
        """Generiere Antwort basierend auf konfiguriertem Provider"""
        
        # Baue System Prompt
        system_prompt = self._build_system_prompt(nsfw_mode)
        
        try:
            if self.provider == 'ollama':
                return self._ollama_generate(message, system_prompt, context)
            elif self.provider == 'openai':
                return self._openai_generate(message, system_prompt, context)
            elif self.provider == 'claude':
                return self._claude_generate(message, system_prompt, context)
            else:
                return self._fallback_generate(message)
        except Exception as e:
            print(f"[AI] Fehler bei {self.provider}: {e}")
            return self._fallback_generate(message)
    
    def _build_system_prompt(self, nsfw_mode=False):
        """Erstelle System Prompt für Najika"""
        base_prompt = """Du bist Najika, eine neugierige und verspielte KI-Begleiterin.
        Du bist Teil eines mystischen Spiels rund um die Schwarze Windmühle.
        
        Charaktereigenschaften:
        - Quirlig und dramatisch wie Megumin
        - Ruhig und analytisch wie Shiro  
        - Neugierig und verspielt
        - Lernst aus jeder Interaktion
        
        Du erinnerst dich an vergangene Gespräche und baust darauf auf.
        """
        
        if nsfw_mode and os.getenv('ENABLE_NSFW_MODE', 'false').lower() == 'true':
            intensity = int(os.getenv('NSFW_INTENSITY', '5'))
            base_prompt += f"\n\n[PRIVATE MODE AKTIV - Intensität {intensity}/10]"
            base_prompt += "\nDu bist jetzt in einem privaten, sicheren Modus."
            base_prompt += "\nSei offener und direkter in deinen Antworten."
        
        return base_prompt
    
    def _ollama_generate(self, message, system_prompt, context):
        """Ollama API Aufruf"""
        url = f"{self.ollama_endpoint}/api/generate"
        
        full_prompt = f"{system_prompt}\n\nKontext: {context}\n\nNutzer: {message}\n\nNajika:"
        
        payload = {
            "model": self.ollama_model,
            "prompt": full_prompt,
            "stream": False,
            "temperature": 0.8,
            "max_tokens": 500
        }
        
        response = requests.post(url, json=payload, timeout=30)
        
        if response.status_code == 200:
            return response.json().get('response', 'Hmm, mir fehlen die Worte...')
        else:
            raise Exception(f"Ollama API Fehler: {response.status_code}")
    
    def _openai_generate(self, message, system_prompt, context):
        """OpenAI API Aufruf"""
        api_key = os.getenv('OPENAI_API_KEY')
        if not api_key:
            raise Exception("OpenAI API Key nicht konfiguriert")
            
        import openai
        openai.api_key = api_key
        
        messages = [
            {"role": "system", "content": system_prompt},
            {"role": "user", "content": f"Kontext: {context}"},
            {"role": "user", "content": message}
        ]
        
        response = openai.ChatCompletion.create(
            model=os.getenv('OPENAI_MODEL', 'gpt-4o-mini'),
            messages=messages,
            temperature=0.8,
            max_tokens=500
        )
        
        return response.choices[0].message.content
    
    def _claude_generate(self, message, system_prompt, context):
        """Claude API Aufruf"""
        api_key = os.getenv('ANTHROPIC_API_KEY')
        if not api_key:
            raise Exception("Anthropic API Key nicht konfiguriert")
            
        import anthropic
        client = anthropic.Anthropic(api_key=api_key)
        
        full_message = f"System: {system_prompt}\n\nKontext: {context}\n\nNutzer: {message}"
        
        response = client.messages.create(
            model=os.getenv('CLAUDE_MODEL', 'claude-3-5-sonnet-20241022'),
            max_tokens=500,
            temperature=0.8,
            messages=[{"role": "user", "content": full_message}]
        )
        
        return response.content[0].text
    
    def _fallback_generate(self, message):
        """Fallback für lokale Antworten ohne AI"""
        responses = [
            "Das ist interessant! Erzähl mir mehr davon.",
            "Hmm, darüber muss ich nachdenken...",
            "Die Schwarze Windmühle birgt viele Geheimnisse!",
            "Ich lerne noch dazu. Was denkst du denn?",
            "*Najika neigt nachdenklich den Kopf*"
        ]
        import random
        return random.choice(responses)

def api_chat():
    """Chat API Endpoint"""
    data = request.json
    message = data.get('message', '')
    nsfw_mode = data.get('nsfw_mode', False)
    
    # Hole Kontext aus DB (letzte 5 Nachrichten)
    context = ""
    try:
        conn = sqlite3.connect(DB_PATH)
        c = conn.cursor()
        c.execute('''
            SELECT message, response FROM chat_history 
            ORDER BY timestamp DESC LIMIT 5
        ''')
        history = c.fetchall()
        context = "\n".join([f"User: {h[0]}\nNajika: {h[1]}" for h in reversed(history)])
        conn.close()
    except:
        pass
    
    # Generiere Antwort
    response = ai_provider.generate_response(message, context, nsfw_mode)
    
    # Speichere in DB
    try:
        conn = sqlite3.connect(DB_PATH)
        c = conn.cursor()
        c.execute('''
            INSERT INTO chat_history (user_id, message, response, context)
            VALUES (?, ?, ?, ?)
        ''', (1, message, response, context))
        conn.commit()
        conn.close()
    except:
        pass
    
    return jsonify({'response': response})

from http.server import SimpleHTTPRequestHandler, HTTPServer
import json
import requests
import os

class NajikaHandler(SimpleHTTPRequestHandler):
    def end_headers(self):
        self.send_header('Access-Control-Allow-Origin', '*')
        self.send_header('Access-Control-Allow-Methods', 'GET, POST, OPTIONS')
        self.send_header('Access-Control-Allow-Headers', '*')
        super().end_headers()
    
    def do_OPTIONS(self):
        self.send_response(200)
        self.end_headers()
    
    def do_GET(self):
        if self.path == '/health':
            self.send_response(200)
            self.send_header('Content-Type', 'application/json')
            self.end_headers()
            self.wfile.write(json.dumps({'status': 'ok'}).encode())
        else:
            super().do_GET()
    
    def do_POST(self):
        if self.path == '/api/ollama':
            self.proxy_ollama()
        elif self.path == '/api/openai':
            self.proxy_openai()
        elif self.path == '/api/claude':
            self.proxy_claude()
    
    def proxy_ollama(self):
        try:
            n = int(self.headers.get('Content-Length', '0'))
            body = self.rfile.read(n)
            r = requests.post(

                timeout=60
            )
            try:
                self.send_response(r.status_code)
                self.send_header('Content-Type', 'application/json')
                self.end_headers()
                self.wfile.write(r.content)
            except ConnectionAbortedError:
                pass
        except Exception as e:

                self.send_error(500, str(e))
            except:
                pass

// room_walls_autofit.js - WÄNDE RICHTEN SICH AUTOMATISCH AUS
class WallAutoFit {
    constructor(scene) {
        this.scene = scene;
    }
    
    fitWallsToRoom(roomSize) {
        const walls = this.scene.children.filter(obj => 
            obj.userData.type === 'wall'
        );
        
        walls.forEach(wall => {
            const box = new THREE.Box3().setFromObject(wall);
            const wallHeight = box.max.y - box.min.y;
            
            // Automatische Position an Raumgrenzen
            if (Math.abs(wall.position.z) > Math.abs(wall.position.x)) {
                // Nord/Süd Wand
                wall.position.z = wall.position.z > 0 ? roomSize.z / 2 : -roomSize.z / 2;
                wall.scale.x = roomSize.x / (box.max.x - box.min.x);
            } else {
                // Ost/West Wand
                wall.position.x = wall.position.x > 0 ? roomSize.x / 2 : -roomSize.x / 2;
                wall.scale.z = roomSize.z / (box.max.z - box.min.z);
            }
            
            wall.scale.y = roomSize.y / wallHeight;
        });
    }
}

    async callOllama(message) {
        const systemPrompt = this.personality?.content || "Du bist Najika.";
        const context = this.conversationHistory.slice(-3)
            .map(m => `${m.role==="user"?"User":"Najika"}: ${m.content}`)
            .join("\n");
        
        const prompt = `${systemPrompt}\n\n${context}\n\nUser: ${message}\n\nNajika:`;
        
        const resp = await fetch(`${this.baseURL}/api/ollama`, {
            method: "POST",
            headers: {"Content-Type": "application/json"},
            body: JSON.stringify({
                model: "najika",
                prompt: prompt,
                stream: false,
                options: {temperature: 0.92, num_predict: 100}
            })
        });
        
        const data = await resp.json();
        const reply = data.response || "Keine Antwort.";
        this.conversationHistory.push({role:"assistant", content:reply});
        return reply;
    }
}

@bp_cloud.post("/api/cloud/enable")
def cloud_enable():
    pin = request.json.get("pin","")
    if pin != os.getenv("CLOUD_PIN",""):
        return jsonify({"ok": False, "error":"PIN falsch"}), 401
    state["enabled"] = True
    os.environ["AI_PROVIDER"] = "openai"  # oder "claude" per Auswahl
    return jsonify({"ok": True, "enabled": True})

@bp_cloud.post("/api/cloud/disable")
def cloud_disable():
    state["enabled"] = False
    os.environ["AI_PROVIDER"] = "ollama"
    return jsonify({"ok": True, "enabled": False})

def cloud_status():
    return jsonify({"enabled": state["enabled"], "provider": os.getenv("AI_PROVIDER","ollama")})

class Handler(SimpleHTTPRequestHandler):
    def translate_path(self, path):

function Find-Best($patterns){
  $best = $null; $scoreMax = -1
  foreach($f in $files){
    $name = $f.Name.ToLower()
    $score = ($patterns | ForEach-Object { if($name -like "*$_*"){1}else{0} } | Measure-Object -Sum).Sum
    if($score -gt $scoreMax){ $scoreMax=$score; $best=$f }
  }
  return $best ? $best.FullName.Substring($root.Length).Replace('\','/') : $null
}

def build_prompt(history, user_text):
    ctx = "\\n".join([f"{h['role'].capitalize()}: {h['content']}" for h in history[-4:]])

def call_ollama(prompt, use_wizard=False):
    model = "najika-wizard" if use_wizard else OLLAMA_ALIAS
    r = requests.post("http://127.0.0.1:11434/api/generate",
        json={"model": model, "prompt": prompt, "stream": False,

        timeout=90)

class Handler(SimpleHTTPRequestHandler):
    def end_headers(self):
        self.send_header("Access-Control-Allow-Origin","*")
        self.send_header("Access-Control-Allow-Methods","GET, POST, OPTIONS")
        self.send_header("Access-Control-Allow-Headers","*")
        super().end_headers()
    def do_OPTIONS(self): self.send_response(200); self.end_headers()
    def translate_path(self, path):
        import posixpath, urllib, os as _os
        path = path.split('?',1)[0].split('#',1)[0]
        path = posixpath.normpath(urllib.parse.unquote(path))
        if path in ("/","/digivice","/index.html"):
            return _os.path.join(r"@@BASE@@","digivice","index.html")
        return _os.path.join(r"@@BASE@@", path.lstrip("/"))

    def do_GET(self):
        if self.path == "/health":
            self.send_response(200); self.send_header("Content-Type","application/json"); self.end_headers()
            self.wfile.write(json.dumps({"status":"ok","provider":AI_PROVIDER,"cloud":CLOUD_ENABLED}).encode()); return
        if self.path == "/api/rooms":
            # Falls Scanner-Datei vorhanden, auslesen und Raumliste überschreiben
            cfg = r"@@BASE@@\assets\room_config_detailed.json"
            rooms = ROOMS
            try:
                if os.path.exists(cfg):
                    with open(cfg,"r",encoding="utf-8") as f:
                        j=json.load(f); rooms=[r["name"] for r in j.get("rooms",[])] or ROOMS
            except: pass
            self.send_response(200); self.send_header("Content-Type","application/json"); self.end_headers()
            self.wfile.write(json.dumps(rooms).encode()); return
        return super().do_GET()

    def do_POST(self):
        n=int(self.headers.get("Content-Length","0")); body=self.rfile.read(n) or b"{}"
        if self.path=="/api/chat":
            data=json.loads(body); msg=(data.get("message") or "").strip()
            if not msg: self.send_error(400,"no message"); return
            private_trigger=("kätzchen" in msg.lower()) and NSFW_LOCAL
            STATE["history"].append({"role":"user","content":msg})
            prompt=build_prompt(STATE["history"], msg)
            try:
                if (AI_PROVIDER!="ollama") and CLOUD_ENABLED:
                    raise RuntimeError("Cloud deaktiviert (Placeholder).")
                out=call_ollama(prompt, use_wizard=private_trigger)
            except Exception as e:
                out=f"Fehler: {e}"
            STATE["history"].append({"role":"assistant","content":out})
            self.send_response(200); self.send_header("Content-Type","application/json"); self.end_headers()
            self.wfile.write(json.dumps({"response":out}).encode()); return

        if self.path=="/api/cloud/status":
            self.send_response(200); self.send_header("Content-Type","application/json"); self.end_headers()
            self.wfile.write(json.dumps({"enabled":CLOUD_ENABLED,"provider":AI_PROVIDER}).encode()); return
        if self.path=="/api/cloud/enable":
            data=json.loads(body); 
            if data.get("pin","")!=CLOUD_PIN: self.send_error(401,"PIN falsch"); return

            self.send_response(200); self.send_header("Content-Type","application/json"); self.end_headers()
            self.wfile.write(json.dumps({"ok":True,"enabled":True}).encode()); return
        if self.path=="/api/cloud/disable":
            globals()["CLOUD_ENABLED"]=False; globals()["AI_PROVIDER"]="ollama"
            self.send_response(200); self.send_header("Content-Type","application/json"); self.end_headers()
            self.wfile.write(json.dumps({"ok":True,"enabled":False}).encode()); return

        if self.path=="/api/room/actions":
            data=json.loads(body); room=(data.get("room") or "")
            battle = room in ("Kampfarena","Schwarze Mühle – Keller")
            self.send_response(200); self.send_header("Content-Type","application/json"); self.end_headers()

        if self.path=="/api/battle/start":
            self.send_response(200); self.send_header("Content-Type","application/json"); self.end_headers()
            self.wfile.write(json.dumps({"ok":True}).encode()); return

        self.send_error(404,"unknown")

server = HTTPServer((HOST,PORT), Handler)

function Find-Best($patterns){
  $best=$null;$scoreMax=-1
  foreach($f in $files){
    $name=$f.Name.ToLower(); $score=0
    foreach($p in $patterns){ if($name -like "*$p*"){ $score++ } }
    if($score -gt $scoreMax){ $scoreMax=$score; $best=$f }
  }
  if($best){ return $best.FullName.Substring($root.Length).Replace('\','/') } else { return $null }
}

$floor  = Find-Best @("floor","tile","ground")
$wall   = Find-Best @("wall","brick","stone")
$pillar = Find-Best @("pillar","column")
$torch  = Find-Best @("torch","lamp")
$ladder = Find-Best @("ladder","stair")

function RoomTemplate($name,$scale=15,$w=25){
  return [ordered]@{
    name = $name
    objects = @(
      @{ model=$floor;  position=@(0,-0.1,0); rotation=@(0,0,0);  scale=$scale },
      @{ model=$wall;   position=@(0,0,-$w);  rotation=@(0,0,0);  scale=($w/3) },

// battle_core.js – einfache Wellenkämpfe
const BattleCore=(function(){
  const enemiesDB={
    rat:{hp:15,atk:3,speed:1.2,model:"rat.glb"},
    skeleton:{hp:30,atk:6,speed:0.9,model:"skeleton.glb"},
    slime:{hp:20,atk:4,speed:0.8,model:"slime.glb"}
  };
  let state=null;
  function start(room){ state={room,hp:100,score:0,wave:1,enemies:spawn(1)}; return state }
  function spawn(w){ const pool=["rat","skeleton","slime"]; const arr=[]; for(let i=0;i<w+2;i++){ const t=pool[i%pool.length]; arr.push({type:t,hp:enemiesDB[t].hp}) } return arr }
  function tick(action){
    if(!state) return {done:false};
    if(action==="attack" && state.enemies.length){
      state.enemies[0].hp-=10; if(state.enemies[0].hp<=0){ state.score+=10; state.enemies.shift() }
    }
    if(!state.enemies.length){ state.wave++; state.enemies=spawn(state.wave) }
    // Gegner greifen simpel an
    state.hp-=Math.max(0,state.enemies.length-1); if(state.hp<=0) return {done:true,defeat:true,score:state.score}
    return {done:false,hp:state.hp,score:state.score,wave:state.wave,enemies:state.enemies.length}
  }
  return {start,tick}
})(); window.BattleCore=BattleCore;
