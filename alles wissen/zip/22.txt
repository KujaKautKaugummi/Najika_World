$files = @(
  "C:\Najika\hub\server.js",

foreach($f in $files){
  if(Test-Path $f){
    $bytes = [IO.File]::ReadAllBytes($f)
    if($bytes.Length -ge 3 -and $bytes[0] -eq 0xEF -and $bytes[1] -eq 0xBB -and $bytes[2] -eq 0xBF){
      [IO.File]::WriteAllBytes($f, $bytes[3..($bytes.Length-1)])
      Write-Host "BOM entfernt: $f" -ForegroundColor Yellow
    } else {
      Write-Host "OK (kein BOM): $f" -ForegroundColor Green
    }
  }
}

function MakeBg([string]$path,[string]$title){
  $w=1920;$h=1080
  $bmp = New-Object Drawing.Bitmap($w,$h)
  $g = [Drawing.Graphics]::FromImage($bmp)
  $rect = New-Object Drawing.Rectangle(0,0,$w,$h)
  $c1=[Drawing.Color]::FromArgb(255,26,33,62)
  $c2=[Drawing.Color]::FromArgb(255,118,75,162)
  $br = New-Object Drawing.Drawing2D.LinearGradientBrush($rect,$c1,$c2,45)
  $g.FillRectangle($br,$rect)
  $pen = New-Object Drawing.Pen([Drawing.Color]::FromArgb(60,255,255,255),8)
  $g.DrawRectangle($pen,10,10,$w-20,$h-20)
  $fnt = New-Object Drawing.Font("Segoe UI",48,[Drawing.FontStyle]::Bold)
  $sf = New-Object Drawing.StringFormat
  $sf.Alignment=[Drawing.StringAlignment]::Center
  $sf.LineAlignment=[Drawing.StringAlignment]::Center
  $shadow = New-Object Drawing.SolidBrush([Drawing.Color]::FromArgb(150,0,0,0))
  $text   = New-Object Drawing.SolidBrush([Drawing.Color]::FromArgb(230,255,255,255))
  $g.DrawString($title,$fnt,$shadow, ($w/2)+3, ($h/2)+3, $sf)
  $g.DrawString($title,$fnt,$text,   $w/2,     $h/2,     $sf)
  $codec=[Drawing.Imaging.ImageCodecInfo]::GetImageEncoders()|?{$_.MimeType -eq "image/jpeg"}
  $encParams = New-Object Drawing.Imaging.EncoderParameters(1)
  $encParams.Param[0]=New-Object Drawing.Imaging.EncoderParameter([Drawing.Imaging.Encoder]::Quality,90L)
  $bmp.Save($path,$codec,$encParams)
  $g.Dispose();$bmp.Dispose();$br.Dispose();$pen.Dispose();$fnt.Dispose();$sf.Dispose();$shadow.Dispose();$text.Dispose()
}
