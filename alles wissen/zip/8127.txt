function setServerStatus(online) {
    if (!statusDot || !statusText) return;
    if (online) {
        statusDot.classList.remove('"'"'offline'"'"');
        statusText.textContent = '"'"'Online'"'"';
    } else {
        statusDot.classList.add('"'"'offline'"'"');
        statusText.textContent = '"'"'Offline'"'"';
    }
}

window.setServerStatus = setServerStatus;

async function api(path, method = '"'"'GET'"'"', body) {
    const options = {
        method,
        headers: {'"'"'Content-Type'"'"': '"'"'application/json'"'"'}
    };
    if (body) {
        options.body = JSON.stringify(body);
    }
    try {
        const response = await fetch(path, options);
        if (!response.ok) {
            throw new Error(response.status + '"'"' '"'"' + response.statusText);
        }
        const data = await response.json();
        setServerStatus(true);
        return data;
    } catch (error) {
        setServerStatus(false);
        throw error;
    }
}

async function toggleCloud() {
    try {

            if (!pin) return;

        console.warn('"'"'Cloud-Umschaltung fehlgeschlagen.'"'"', error);

async function renderActions(room) {
    let resp;
    try {
        resp = await api('"'"'/api/room/actions'"'"', '"'"'POST'"'"', {room});
    } catch (error) {
        console.warn('"'"'Nutze Fallback-Aktionen fr "'"'"' + room + '"'"'".'"'"', error);
        resp = {
            battle: BATTLE_ROOMS.has(room),
            actions: ROOM_ACTIONS_FALLBACK[room] || []
        };
    }
    const wrap = document.getElementById('"'"'room-actions'"'"');
    wrap.innerHTML = '"''"';
    const actions = resp && resp.actions ? resp.actions : ROOM_ACTIONS_FALLBACK[room] || [];
    actions.forEach(a => {
        const b = document.createElement('"'"'button'"'"');
        b.className = '"'"'action-btn'"'"';
        b.textContent = a;
        b.onclick = () => handleAction(room, a);
        wrap.appendChild(b);
    });
    const battlePanel = document.getElementById('"'"'battlePanel'"'"');
    if (!resp || !resp.battle) {
        battlePanel.style.display = '"'"'none'"'"';
    }
}

}

function updateBattleHUD(data) {
    const hpEl = document.getElementById('"'"'battleHP'"'"');
    const hpMaxEl = document.getElementById('"'"'battleMaxHP'"'"');
    const waveEl = document.getElementById('"'"'battleWave'"'"');
    const enemyEl = document.getElementById('"'"'battleEnemies'"'"');
    const fillEl = document.getElementById('"'"'hpFill'"'"');
    if (!hpEl || !hpMaxEl || !fillEl) return;
    if (data.hp !== undefined) {
        hpEl.textContent = data.hp;
        hpMaxEl.textContent = data.max_hp || 100;
        const percentage = Math.max(0, Math.min(100, (data.hp / (data.max_hp || 100)) * 100));
        fillEl.style.width = percentage + '"'"'%'"'"';
        if (percentage > 60) {
            fillEl.style.background = '"'"'linear-gradient(90deg, #2ecc71, #27ae60)'"'"';
        } else if (percentage > 30) {
            fillEl.style.background = '"'"'linear-gradient(90deg, #f39c12, #e67e22)'"'"';
        } else {
            fillEl.style.background = '"'"'linear-gradient(90deg, #e74c3c, #c0392b)'"'"';
        }
    }
    if (waveEl && data.wave !== undefined) {
        waveEl.textContent = data.wave;
    }
    if (enemyEl && data.enemies !== undefined) {
        enemyEl.textContent = data.enemies;
    }
}
