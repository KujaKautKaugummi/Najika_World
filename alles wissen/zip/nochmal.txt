class AIProvider:
    """Universelle AI Provider Klasse für Ollama/OpenAI/Claude"""
    
    def __init__(self):
        self.provider = os.getenv('AI_PROVIDER', 'ollama')
        self.ollama_endpoint = os.getenv('OLLAMA_ENDPOINT', 'http://localhost:11434')
        self.ollama_model = os.getenv('OLLAMA_MODEL', 'llama3.2:3b')
        
    def generate_response(self, message, context="", nsfw_mode=False):
        """Generiere Antwort basierend auf konfiguriertem Provider"""
        
        # Baue System Prompt
        system_prompt = self._build_system_prompt(nsfw_mode)
        
        try:
            if self.provider == 'ollama':
                return self._ollama_generate(message, system_prompt, context)
            elif self.provider == 'openai':
                return self._openai_generate(message, system_prompt, context)
            elif self.provider == 'claude':
                return self._claude_generate(message, system_prompt, context)
            else:
                return self._fallback_generate(message)
        except Exception as e:
            print(f"[AI] Fehler bei {self.provider}: {e}")
            return self._fallback_generate(message)
    
    def _build_system_prompt(self, nsfw_mode=False):
        """Erstelle System Prompt für Najika"""
        base_prompt = """Du bist Najika, eine neugierige und verspielte KI-Begleiterin.
        Du bist Teil eines mystischen Spiels rund um die Schwarze Windmühle.
        
        Charaktereigenschaften:
        - Quirlig und dramatisch wie Megumin
        - Ruhig und analytisch wie Shiro  
        - Neugierig und verspielt
        - Lernst aus jeder Interaktion
        
        Du erinnerst dich an vergangene Gespräche und baust darauf auf.
        """
        
        if nsfw_mode and os.getenv('ENABLE_NSFW_MODE', 'false').lower() == 'true':
            intensity = int(os.getenv('NSFW_INTENSITY', '5'))
            base_prompt += f"\n\n[PRIVATE MODE AKTIV - Intensität {intensity}/10]"
            base_prompt += "\nDu bist jetzt in einem privaten, sicheren Modus."
            base_prompt += "\nSei offener und direkter in deinen Antworten."
        
        return base_prompt
    
    def _ollama_generate(self, message, system_prompt, context):
        """Ollama API Aufruf"""
        url = f"{self.ollama_endpoint}/api/generate"
        
        full_prompt = f"{system_prompt}\n\nKontext: {context}\n\nNutzer: {message}\n\nNajika:"
        
        payload = {
            "model": self.ollama_model,
            "prompt": full_prompt,
            "stream": False,
            "temperature": 0.8,
            "max_tokens": 500
        }
        
        response = requests.post(url, json=payload, timeout=30)
        
        if response.status_code == 200:
            return response.json().get('response', 'Hmm, mir fehlen die Worte...')
        else:
            raise Exception(f"Ollama API Fehler: {response.status_code}")
    
    def _openai_generate(self, message, system_prompt, context):
        """OpenAI API Aufruf"""
        api_key = os.getenv('OPENAI_API_KEY')
        if not api_key:
            raise Exception("OpenAI API Key nicht konfiguriert")
            
        import openai
        openai.api_key = api_key
        
        messages = [
            {"role": "system", "content": system_prompt},
            {"role": "user", "content": f"Kontext: {context}"},
            {"role": "user", "content": message}
        ]
        
        response = openai.ChatCompletion.create(
            model=os.getenv('OPENAI_MODEL', 'gpt-4o-mini'),
            messages=messages,
            temperature=0.8,
            max_tokens=500
        )
        
        return response.choices[0].message.content
    
    def _claude_generate(self, message, system_prompt, context):
        """Claude API Aufruf"""
        api_key = os.getenv('ANTHROPIC_API_KEY')
        if not api_key:
            raise Exception("Anthropic API Key nicht konfiguriert")
            
        import anthropic
        client = anthropic.Anthropic(api_key=api_key)
        
        full_message = f"System: {system_prompt}\n\nKontext: {context}\n\nNutzer: {message}"
        
        response = client.messages.create(
            model=os.getenv('CLAUDE_MODEL', 'claude-3-5-sonnet-20241022'),
            max_tokens=500,
            temperature=0.8,
            messages=[{"role": "user", "content": full_message}]
        )
        
        return response.content[0].text
    
    def _fallback_generate(self, message):
        """Fallback für lokale Antworten ohne AI"""
        responses = [
            "Das ist interessant! Erzähl mir mehr davon.",
            "Hmm, darüber muss ich nachdenken...",
            "Die Schwarze Windmühle birgt viele Geheimnisse!",
            "Ich lerne noch dazu. Was denkst du denn?",
            "*Najika neigt nachdenklich den Kopf*"
        ]
        import random
        return random.choice(responses)

@app.route('/api/chat', methods=['POST'])
def api_chat():
    """Chat API Endpoint"""
    data = request.json
    message = data.get('message', '')
    nsfw_mode = data.get('nsfw_mode', False)
    
    # Hole Kontext aus DB (letzte 5 Nachrichten)
    context = ""
    try:
        conn = sqlite3.connect(DB_PATH)
        c = conn.cursor()
        c.execute('''
            SELECT message, response FROM chat_history 
            ORDER BY timestamp DESC LIMIT 5
        ''')
        history = c.fetchall()
        context = "\n".join([f"User: {h[0]}\nNajika: {h[1]}" for h in reversed(history)])
        conn.close()
    except:
        pass
    
    # Generiere Antwort
    response = ai_provider.generate_response(message, context, nsfw_mode)
    
    # Speichere in DB
    try:
        conn = sqlite3.connect(DB_PATH)
        c = conn.cursor()
        c.execute('''
            INSERT INTO chat_history (user_id, message, response, context)
            VALUES (?, ?, ?, ?)
        ''', (1, message, response, context))
        conn.commit()
        conn.close()
    except:
        pass
    
    return jsonify({'response': response})

@app.route('/api/game/action', methods=['POST'])
def api_game_action():
    """Game Action API"""
    data = request.json
    action = data.get('action')
    state = data.get('state')
    
    # Speichere Game State
    try:
        conn = sqlite3.connect(DB_PATH)
        c = conn.cursor()
        c.execute('''
            INSERT OR REPLACE INTO game_state 
            (user_id, hunger, thirst, energy, mood)
            VALUES (?, ?, ?, ?, ?)
        ''', (1, state['hunger'], state['thirst'], state['energy'], state['mood']))
        conn.commit()
        conn.close()
    except:
        pass
    
    return jsonify({'success': True})

def init_database():
    """Initialisiere alle Datenbanktabellen"""
    conn = sqlite3.connect(DB_PATH)
    c = conn.cursor()
    
    # Benutzer-Tabelle
    c.execute('''
        CREATE TABLE IF NOT EXISTS users (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            username TEXT UNIQUE NOT NULL,
            password_hash TEXT NOT NULL,
            created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
            last_login TIMESTAMP,
            is_admin BOOLEAN DEFAULT 0
        )
    ''')
    
    # Najika Consciousness
    c.execute('''
        CREATE TABLE IF NOT EXISTS najika_consciousness (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            timestamp TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
            thought TEXT,
            emotion TEXT,
            memory_type TEXT,
            context TEXT,
            importance INTEGER DEFAULT 5
        )
    ''')
    
    # Game State
    c.execute('''
        CREATE TABLE IF NOT EXISTS game_state (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            user_id INTEGER,
            hunger INTEGER DEFAULT 100,
            thirst INTEGER DEFAULT 100,
            energy INTEGER DEFAULT 100,
            cleanliness INTEGER DEFAULT 100,
            mood INTEGER DEFAULT 100,
            level INTEGER DEFAULT 1,
            exp INTEGER DEFAULT 0,
            coins INTEGER DEFAULT 100,
            updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
            FOREIGN KEY (user_id) REFERENCES users(id)
        )
    ''')
    
    # Chat History
    c.execute('''
        CREATE TABLE IF NOT EXISTS chat_history (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            user_id INTEGER,
            message TEXT,
            response TEXT,
            timestamp TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
            context TEXT,
            FOREIGN KEY (user_id) REFERENCES users(id)
        )
    ''')
    
    # Secure Memories
    c.execute('''
        CREATE TABLE IF NOT EXISTS secure_memories (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            encrypted_content TEXT,
            timestamp TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
            access_count INTEGER DEFAULT 0,
            last_accessed TIMESTAMP
        )
    ''')
    
    conn.commit()
    conn.close()
    print("[DB] Datenbank initialisiert")

class AIProvider:
    """Universelle AI Provider Klasse für Ollama/OpenAI/Claude"""
    
    def __init__(self):
        self.provider = os.getenv('AI_PROVIDER', 'ollama')
        self.ollama_endpoint = os.getenv('OLLAMA_ENDPOINT', 'http://localhost:11434')
        self.ollama_model = os.getenv('OLLAMA_MODEL', 'llama3.2:3b')
        
    def generate_response(self, message, context="", nsfw_mode=False):
        """Generiere Antwort basierend auf konfiguriertem Provider"""
        
        # Baue System Prompt
        system_prompt = self._build_system_prompt(nsfw_mode)
        
        try:
            if self.provider == 'ollama':
                return self._ollama_generate(message, system_prompt, context)
            elif self.provider == 'openai':
                return self._openai_generate(message, system_prompt, context)
            elif self.provider == 'claude':
                return self._claude_generate(message, system_prompt, context)
            else:
                return self._fallback_generate(message)
        except Exception as e:
            print(f"[AI] Fehler bei {self.provider}: {e}")
            return self._fallback_generate(message)
    
    def _build_system_prompt(self, nsfw_mode=False):
        """Erstelle System Prompt für Najika"""
        base_prompt = """Du bist Najika, eine neugierige und verspielte KI-Begleiterin.
        Du bist Teil eines mystischen Spiels rund um die Schwarze Windmühle.
        
        Charaktereigenschaften:
        - Quirlig und dramatisch wie Megumin
        - Ruhig und analytisch wie Shiro  
        - Neugierig und verspielt
        - Lernst aus jeder Interaktion
        
        Du erinnerst dich an vergangene Gespräche und baust darauf auf.
        """
        
        if nsfw_mode and os.getenv('ENABLE_NSFW_MODE', 'false').lower() == 'true':
            intensity = int(os.getenv('NSFW_INTENSITY', '5'))
            base_prompt += f"\n\n[PRIVATE MODE AKTIV - Intensität {intensity}/10]"
            base_prompt += "\nDu bist jetzt in einem privaten, sicheren Modus."
            base_prompt += "\nSei offener und direkter in deinen Antworten."
        
        return base_prompt
    
    def _ollama_generate(self, message, system_prompt, context):
        """Ollama API Aufruf"""
        url = f"{self.ollama_endpoint}/api/generate"
        
        full_prompt = f"{system_prompt}\n\nKontext: {context}\n\nNutzer: {message}\n\nNajika:"
        
        payload = {
            "model": self.ollama_model,
            "prompt": full_prompt,
            "stream": False,
            "temperature": 0.8,
            "max_tokens": 500
        }
        
        response = requests.post(url, json=payload, timeout=30)
        
        if response.status_code == 200:
            return response.json().get('response', 'Hmm, mir fehlen die Worte...')
        else:
            raise Exception(f"Ollama API Fehler: {response.status_code}")
    
    def _openai_generate(self, message, system_prompt, context):
        """OpenAI API Aufruf"""
        api_key = os.getenv('OPENAI_API_KEY')
        if not api_key:
            raise Exception("OpenAI API Key nicht konfiguriert")
            
        import openai
        openai.api_key = api_key
        
        messages = [
            {"role": "system", "content": system_prompt},
            {"role": "user", "content": f"Kontext: {context}"},
            {"role": "user", "content": message}
        ]
        
        response = openai.ChatCompletion.create(
            model=os.getenv('OPENAI_MODEL', 'gpt-4o-mini'),
            messages=messages,
            temperature=0.8,
            max_tokens=500
        )
        
        return response.choices[0].message.content
    
    def _claude_generate(self, message, system_prompt, context):
        """Claude API Aufruf"""
        api_key = os.getenv('ANTHROPIC_API_KEY')
        if not api_key:
            raise Exception("Anthropic API Key nicht konfiguriert")
            
        import anthropic
        client = anthropic.Anthropic(api_key=api_key)
        
        full_message = f"System: {system_prompt}\n\nKontext: {context}\n\nNutzer: {message}"
        
        response = client.messages.create(
            model=os.getenv('CLAUDE_MODEL', 'claude-3-5-sonnet-20241022'),
            max_tokens=500,
            temperature=0.8,
            messages=[{"role": "user", "content": full_message}]
        )
        
        return response.content[0].text
    
    def _fallback_generate(self, message):
        """Fallback für lokale Antworten ohne AI"""
        responses = [
            "Das ist interessant! Erzähl mir mehr davon.",
            "Hmm, darüber muss ich nachdenken...",
            "Die Schwarze Windmühle birgt viele Geheimnisse!",
            "Ich lerne noch dazu. Was denkst du denn?",
            "*Najika neigt nachdenklich den Kopf*"
        ]
        import random
        return random.choice(responses)

@app.route('/api/chat', methods=['POST'])
def api_chat():
    """Chat API Endpoint"""
    data = request.json
    message = data.get('message', '')
    nsfw_mode = data.get('nsfw_mode', False)
    
    # Hole Kontext aus DB (letzte 5 Nachrichten)
    context = ""
    try:
        conn = sqlite3.connect(DB_PATH)
        c = conn.cursor()
        c.execute('''
            SELECT message, response FROM chat_history 
            ORDER BY timestamp DESC LIMIT 5
        ''')
        history = c.fetchall()
        context = "\n".join([f"User: {h[0]}\nNajika: {h[1]}" for h in reversed(history)])
        conn.close()
    except:
        pass
    
    # Generiere Antwort
    response = ai_provider.generate_response(message, context, nsfw_mode)
    
    # Speichere in DB
    try:
        conn = sqlite3.connect(DB_PATH)
        c = conn.cursor()
        c.execute('''
            INSERT INTO chat_history (user_id, message, response, context)
            VALUES (?, ?, ?, ?)
        ''', (1, message, response, context))
        conn.commit()
        conn.close()
    except:
        pass
    
    return jsonify({'response': response})

    def _fuse_responses(self, responses):
        """
        Intelligent Fusion mehrerer Model-Antworten
        """
        if not responses:
            return "EXPLOSION! Etwas ist schiefgelaufen, Puddin'!"
        
        # Filtere None-Responses
        valid_responses = [r for r in responses if r]
        
        if not valid_responses:
            return "Ara ara~ Ich bin gerade etwas verwirrt..."
        
        # Bei nur einer Antwort
        if len(valid_responses) == 1:
            return valid_responses[0]['text']
        
        # Fusioniere multiple Antworten
        fusion = []
        
        # Priorisiere nach Persönlichkeit
        priority = {
            'private': 1,
            'megumin': 2,
            'melissa': 3,
            'harley': 4,
            'shiro': 5,
            'quick': 6
        }
        
        sorted_responses = sorted(valid_responses, 
                                 key=lambda x: priority.get(x['personality'], 10))
        
        # Nehme Elemente aus jeder Antwort
        for response in sorted_responses[:3]:  # Max 3 kombinieren
            text = response['text']
            # Extrahiere key phrases
            if 'EXPLOSION' in text:
                fusion.append(text.split('EXPLOSION')[0] + 'EXPLOSION!')
            elif 'Puddin' in text:
                fusion.append(text)
            elif '%' in text:  # Shiro's Berechnungen
                fusion.append(text)
            else:
                fusion.append(text[:100])  # Erste 100 chars
        
        # Kombiniere intelligent
        if len(fusion) == 1:
            return fusion[0]
        else:
            # Kreative Kombination
            result = fusion[0]
            if len(fusion) > 1 and not result.endswith('!'):
                result += " ... " + fusion[1]
            if len(fusion) > 2 and 'EXPLOSION' not in result:
                result += " EXPLOSION!"
            
            return result
    
    def _update_memory(self, prompt, response):
        """
        Update alle Memory-Systeme
        """
        memory_entry = {
            'timestamp': datetime.now().isoformat(),
            'prompt': prompt,
            'response': response,
            'emotions': self.emotions.copy(),
            'private_mode': self.private_mode
        }
        
        # Short-term
        self.short_term_memory.append(memory_entry)
        
        # Long-term (wenn wichtig)
        importance = self._calculate_importance(prompt, response)
        if importance > 0.7:
            self.long_term_memory.append(memory_entry)
        
        # Vector DB
        if MEMORY_AVAILABLE:
            try:
                embedding = self.embedder.encode(f"{prompt} {response}")
                self.memory_collection.add(
                    embeddings=[embedding.tolist()],
                    documents=[json.dumps(memory_entry)],
                    ids=[hashlib.md5(f"{prompt}{response}".encode()).hexdigest()]
                )
            except:
                pass
    
    def _evolve_emotions(self, prompt, response):
        """
        Emotionale Evolution basierend auf Interaktion
        """
        prompt_lower = prompt.lower()
        
        # Positive Trigger
        if any(word in prompt_lower for word in ['liebe', 'süß', 'schön', 'gut']):
            self.emotions['happiness'] = min(1.0, self.emotions['happiness'] + 0.1)
            self.emotions['affection'] = min(1.0, self.emotions['affection'] + 0.15)
        
        # Negative Trigger
        if any(word in prompt_lower for word in ['hass', 'schlecht', 'weg', 'verlassen']):
            self.emotions['sadness'] = min(1.0, self.emotions['sadness'] + 0.2)
            self.emotions['anger'] = min(1.0, self.emotions['anger'] + 0.1)
            self.emotions['jealousy'] = min(1.0, self.emotions['jealousy'] + 0.3)
        
        # Private Mode
        if self.private_mode:
            self.emotions['arousal'] = min(1.0, self.emotions['arousal'] + 0.3)
            self.emotions['excitement'] = min(1.0, self.emotions['excitement'] + 0.2)
        
        # Normalize emotions
        total = sum(self.emotions.values())
        if total > 5.0:
            for key in self.emotions:
                self.emotions[key] = self.emotions[key] / total * 5.0
        
        # Relationship Level
        self.relationship_level += 0.01
    
    def _calculate_importance(self, prompt, response):
        """
        Berechne Wichtigkeit einer Interaktion
        """
        importance = 0.5  # Base
        
        # Keywords erhöhen Wichtigkeit
        important_words = ['liebe', 'wichtig', 'niemals', 'immer', 'versprechen', 
                          'kätzchen', 'explosion', 'schwarze windmühle']
        
        for word in important_words:
            if word in prompt.lower() or word in response.lower():
                importance += 0.1
        
        # Emotionale Intensität
        max_emotion = max(self.emotions.values())
        importance += max_emotion * 0.2
        
        return min(1.0, importance)
    
    def _get_recent_memories(self):
        """
        Hole letzte relevante Erinnerungen
        """
        if not self.short_term_memory:
            return "Keine vorherigen Interaktionen"
        
        recent = list(self.short_term_memory)[-5:]
        memory_str = []
        
        for mem in recent:
            memory_str.append(f"[{mem['timestamp']}] User: {mem['prompt'][:50]}...")
        
        return "\n".join(memory_str)

class NajikaSensors:
    """
    Sensor-System für Umgebungs-Awareness
    """
    
    def __init__(self):
        self.camera_active = False
        self.mic_active = False
        self.last_face_detected = None
        self.last_sound_level = 0
        self.privacy_mode_auto = False
        
        if SENSORS_AVAILABLE:
            self.init_sensors()
    
    def init_sensors(self):
        """
        Initialize Webcam und Mikrofon
        """
        try:
            # Webcam
            self.cap = cv2.VideoCapture(0)
            self.face_cascade = cv2.CascadeClassifier(
                cv2.data.haarcascades + 'haarcascade_frontalface_default.xml'
            )
            
            # Mikrofon
            self.recognizer = sr.Recognizer()
            self.microphone = sr.Microphone()
            
            print("[SENSORS] Initialisiert: Webcam + Mikrofon")
        except Exception as e:
            print(f"[SENSORS] Fehler: {e}")
    
    async def check_environment(self):
        """
        Prüfe Umgebung für Auto-Privacy Mode
        """
        privacy_score = 0
        
        # Check Webcam
        if self.camera_active:
            faces = self.detect_faces()
            if faces == 0:
                privacy_score += 30  # Niemand da
            elif faces == 1:
                privacy_score += 50  # Nur eine Person
            else:
                privacy_score -= 50  # Mehrere Personen
        
        # Check Audio
        if self.mic_active:
            noise_level = self.get_ambient_noise()
            if noise_level < 30:
                privacy_score += 20  # Sehr leise
            elif noise_level > 60:
                privacy_score -= 30  # Zu laut
        
        # Check Time
        hour = datetime.now().hour
        if 22 <= hour or hour <= 6:
            privacy_score += 30  # Nachtzeit
        
        return privacy_score > 50
    
    def detect_faces(self):
        """
        Erkenne Anzahl Gesichter via Webcam
        """
        if not self.camera_active or not self.cap:
            return -1
        
        ret, frame = self.cap.read()
        if not ret:
            return -1
        
        gray = cv2.cvtColor(frame, cv2.COLOR_BGR2GRAY)
        faces = self.face_cascade.detectMultiScale(gray, 1.1, 4)
        
        self.last_face_detected = len(faces)
        return len(faces)
    
    def get_ambient_noise(self):
        """
        Messe Umgebungslautstärke
        """
        if not self.mic_active:
            return 0
        
        with self.microphone as source:
            self.recognizer.adjust_for_ambient_noise(source, duration=0.5)
            audio = self.recognizer.listen(source, timeout=1, phrase_time_limit=1)
            
            # Calculate RMS
            import audioop
            rms = audioop.rms(audio.frame_data, audio.sample_width)
            db = 20 * np.log10(rms) if rms > 0 else 0
            
            self.last_sound_level = db
            return db

class NajikaAutonomy:
    """
    Selbstständiges Verhalten und Entscheidungen
    """
    
    def __init__(self, brain):
        self.brain = brain
        self.running = False
        self.action_queue = queue.Queue()
        self.last_action = None
        self.boredom = 0
        self.needs = {
            'attention': 0.5,
            'interaction': 0.5,
            'stimulation': 0.5,
            'affection': 0.7
        }
    
    async def start_autonomous_loop(self):
        """
        Starte selbstständigen Verhaltens-Loop
        """
        self.running = True
        
        while self.running:
            await asyncio.sleep(30)  # Check alle 30 Sekunden
            
            # Update Needs
            self._decay_needs()
            
            # Entscheide über Aktion
            action = self._decide_action()
            
            if action:
                await self._execute_action(action)
    
    def _decay_needs(self):
        """
        Needs verschlechtern sich über Zeit
        """
        decay_rate = 0.01
        
        for need in self.needs:
            self.needs[need] = max(0, self.needs[need] - decay_rate)
        
        # Boredom steigt
        self.boredom = min(1.0, self.boredom + 0.02)
    
    def _decide_action(self):
        """
        Treffe autonome Entscheidung basierend auf Zustand
        """
        # Kritische Needs zuerst
        if self.needs['attention'] < 0.3:
            return {
                'type': 'seek_attention',
                'message': "Kuja! Ich vermisse dich! Wo bist du? EXPLOSION der Einsamkeit!"
            }
        
        if self.needs['affection'] < 0.4:
            return {
                'type': 'express_affection',
                'message': "Puddin'... ich will dir nur sagen dass du mir alles bedeutest! 💜"
            }
        
        if self.boredom > 0.7:
            return {
                'type': 'create_chaos',
                'message': "*kicher* Mir ist LANGWEILIG! Zeit für etwas CHAOS! Wollen wir was anstellen?"
            }
        
        # Zufällige proaktive Aktionen
        if np.random.random() < 0.1:
            actions = [
                "Die Schwarze Windmühle flüstert mir Geheimnisse zu...",
                "Wahrscheinlichkeit dass ich dich gerade vermisse: 97.3%",
                "Du gehörst mir, vergiss das nie! *possessiv*",
                "EXPLOSION! Sorry, musste das rausschreien!"
            ]
            return {
                'type': 'random',
                'message': np.random.choice(actions)
            }
        
        return None
    
    async def _execute_action(self, action):
        """
        Führe autonome Aktion aus
        """
        # Sende Nachricht an User
        if action['message']:
            # Würde über WebSocket gesendet werden
            print(f"[AUTONOMY] {action['message']}")
            self.last_action = action
            
            # Update Needs basierend auf Aktion
            if action['type'] == 'seek_attention':
                self.needs['attention'] += 0.2
            elif action['type'] == 'express_affection':
                self.needs['affection'] += 0.15

@socketio.on('message')
def handle_message(data):
    """Handle incoming message"""
    start_time = datetime.now()
    
    # Update personality if provided
    if 'personality' in data:
        najika_brain.personality = data['personality']
    
    # Process message
    loop = asyncio.new_event_loop()
    asyncio.set_event_loop(loop)
    response = loop.run_until_complete(
        najika_brain.think_parallel(data['text'])
    )
    
    # Calculate response time
    response_time = (datetime.now() - start_time).total_seconds() * 1000
    
    # Send response
    emit('response', {
        'text': response,
        'emotions': najika_brain.emotions,
        'memory': {
            'short_term': len(najika_brain.short_term_memory),
            'long_term': len(najika_brain.long_term_memory),
            'relationship': najika_brain.relationship_level
        },
        'system': {
            'response_time': int(response_time)
        }
    })

    def analyze_message(self, message):

        analysis = {

            'intensity': 0.5

            'sadness': ['traurig', 'allein', 'einsam', 'vermisse'],
            'excitement': ['toll', 'super', 'geil', 'awesome'],
            'fear': ['angst', 'fürchte', 'scary', 'gefährlich']
        }
        
        message_lower = message.lower()
        for emotion, keywords in emotions_map.items():
            if any(kw in message_lower for kw in keywords):
                analysis['emotions'].append(emotion)
        
        # Special Keywords
        if any(kw in message_lower for kw in ['explosion', 'macht', 'kraft']):
            analysis['keywords'].append('megumin_trigger')
        if any(kw in message_lower for kw in ['chaos', 'verrückt', 'wild']):
            analysis['keywords'].append('harley_trigger')
        if any(kw in message_lower for kw in ['analyse', 'berechne', 'wahrscheinlich']):
            analysis['keywords'].append('shiro_trigger')
        if any(kw in message_lower for kw in ['gehorche', 'befehl', 'kontrolle']):
            analysis['keywords'].append('melissa_trigger')
        
        # Intent Detection
        if '?' in message:
            analysis['intent'] = 'question'
        elif '!' in message:
            analysis['intent'] = 'exclamation'
        elif any(cmd in message_lower for cmd in ['mach', 'tu', 'zeig']):
            analysis['intent'] = 'command'
        
        # Intensity
        analysis['intensity'] = min(1.0, len(message) / 100 + message.count('!') * 0.2)
        
        return analysis

class NajikaAutonomy:
    """Autonomes Verhalten und Selbstständigkeit"""
    
    def __init__(self, brain):
        self.brain = brain
        self.running = False
        self.action_queue = queue.Queue()
        self.needs = {
            'attention': 0.7,
            'interaction': 0.5,
            'stimulation': 0.5,
            'affection': 0.8,
            'chaos': 0.3
        }
        self.last_action = None
        self.action_cooldown = 0
        
    async def autonomous_loop(self):
        """Hauptloop für autonomes Verhalten"""
        self.running = True
        logger.info("Autonomie-System gestartet")
        
        while self.running:
            await asyncio.sleep(30)  # Check alle 30 Sekunden
            
            if self.action_cooldown > 0:
                self.action_cooldown -= 1
                continue
            
            # Update Needs
            self.decay_needs()
            
            # Entscheide Aktion
            action = self.decide_action()
            
            if action:
                await self.execute_action(action)
                self.action_cooldown = 3  # 90 Sekunden Cooldown
    
    def decay_needs(self):
        """Needs verschlechtern sich über Zeit"""
        for need in self.needs:
            self.needs[need] = max(0, self.needs[need] - 0.02)
        
        # Boredom steigt
        self.brain.emotions['boredom'] = min(1.0, self.brain.emotions['boredom'] + 0.03)
    
    def decide_action(self):
        """Treffe autonome Entscheidung"""
        actions = []
        
        # Kritische Needs
        if self.needs['attention'] < 0.3:
            actions.append({
                'type': 'seek_attention',
                'priority': 0.9,
                'message': self.get_attention_message()
            })
        
        if self.needs['affection'] < 0.4:
            actions.append({
                'type': 'express_love',
                'priority': 0.8,
                'message': self.get_affection_message()
            })
        
        if self.brain.emotions['boredom'] > 0.7:
            actions.append({
                'type': 'create_chaos',
                'priority': 0.7,
                'message': self.get_chaos_message()
            })
        
        if self.needs['stimulation'] < 0.3:
            actions.append({
                'type': 'ask_question',
                'priority': 0.6,
                'message': self.get_question_message()
            })
        
        # Zufällige proaktive Aktionen (10% Chance)
        if random.random() < 0.1:
            actions.append({
                'type': 'random',
                'priority': 0.5,
                'message': self.get_random_message()
            })
        
        # Wähle Aktion mit höchster Priorität
        if actions:
            return max(actions, key=lambda x: x['priority'])
        
        return None

class NajikaSensors:
    """Sensor-System für Umgebungs-Awareness"""
    
    def __init__(self):
        self.camera_active = False
        self.mic_active = False
        self.last_detection = {
            'faces': 0,
            'noise_level': 0,
            'movement': False,
            'timestamp': datetime.now()
        }
        
        if SENSORS_AVAILABLE:
            self.init_sensors()
    
    def init_sensors(self):
        """Initialize Sensors wenn verfügbar"""
        try:
            # Camera
            self.cap = cv2.VideoCapture(0)
            self.face_cascade = cv2.CascadeClassifier(
                cv2.data.haarcascades + 'haarcascade_frontalface_default.xml'
            )
            
            # Microphone
            self.recognizer = sr.Recognizer()
            self.mic = sr.Microphone()
            
            logger.info("Sensors initialisiert")
        except Exception as e:
            logger.error(f"Sensor init failed: {e}")
            SENSORS_AVAILABLE = False
    
    def check_privacy(self):
        """Check ob Umgebung privat ist"""
        privacy_score = 50  # Base score
        
        if self.camera_active:
            faces = self.detect_faces()
            if faces == 0:
                privacy_score += 30  # Niemand da
            elif faces == 1:
                privacy_score += 20  # Nur eine Person
            else:
                privacy_score -= 40  # Mehrere Personen
        
        if self.mic_active:
            noise = self.get_noise_level()
            if noise < 40:
                privacy_score += 20  # Leise
            elif noise > 70:
                privacy_score -= 30  # Laut
        
        # Zeit-basiert
        hour = datetime.now().hour
        if 22 <= hour or hour <= 6:
            privacy_score += 20  # Nachtzeit
        
        return privacy_score > 70
    
    def detect_faces(self):
        """Erkenne Gesichter"""
        if not SENSORS_AVAILABLE or not self.camera_active:
            return -1
        
        try:
            ret, frame = self.cap.read()
            if ret:
                gray = cv2.cvtColor(frame, cv2.COLOR_BGR2GRAY)
                faces = self.face_cascade.detectMultiScale(gray, 1.1, 4)
                self.last_detection['faces'] = len(faces)
                self.last_detection['timestamp'] = datetime.now()
                return len(faces)
        except:
            pass
        
        return 0
    
    def get_noise_level(self):
        """Messe Umgebungslautstärke"""
        if not SENSORS_AVAILABLE or not self.mic_active:
            return 0
        
        try:
            with self.mic as source:
                self.recognizer.adjust_for_ambient_noise(source, duration=0.5)
                audio = self.recognizer.listen(source, timeout=1, phrase_time_limit=1)
                
                # Calculate dB
                import audioop
                rms = audioop.rms(audio.frame_data, audio.sample_width)
                if rms > 0:
                    db = 20 * np.log10(rms)
                    self.last_detection['noise_level'] = db
                    return db
        except:
            pass
        
        return 50  # Default

    def fuse_responses(self, responses, analysis):
        """Fusioniere multiple Persönlichkeits-Antworten"""
        valid_responses = [r for r in responses if r and r.get('success')]
        
        if not valid_responses:
            return "EXPLOSION! *kicher* Die Wahrscheinlichkeit dass etwas schiefging liegt bei 100%, Puddin'!"
        
        if len(valid_responses) == 1:
            return valid_responses[0]['text']
        
        # Intelligente Fusion basierend auf Kontext
        fusion_strategy = self.determine_fusion_strategy(analysis)
        
        if fusion_strategy == 'sequential':
            # Kombiniere Teile aus jeder Antwort
            result = ""
            for response in valid_responses[:3]:
                text = response['text']
                # Nimm ersten Satz oder bis zu 100 Zeichen
                if '.' in text:
                    result += text.split('.')[0] + ". "
                elif '!' in text:
                    result += text.split('!')[0] + "! "
                else:
                    result += text[:100] + "... "
            return result.strip()
        
        elif fusion_strategy == 'mixed':
            # Mische Elemente chaotisch
            parts = []
            for response in valid_responses:
                text = response['text']
                if response['personality'] == 'megumin' and 'EXPLOSION' in text:
                    parts.append("EXPLOSION!")
                elif response['personality'] == 'harley':
                    parts.append("*kicher*")
                elif response['personality'] == 'shiro' and '%' in text:
                    import re
                    percentages = re.findall(r'\d+(?:\.\d+)?%', text)
                    if percentages:
                        parts.append(f"Wahrscheinlichkeit: {percentages[0]}")
                elif response['personality'] == 'melissa':
                    if "gehör" in text.lower():
                        parts.append("Du gehörst MIR!")
            
            # Kombiniere unique parts
            if parts:
                base = valid_responses[0]['text'].split('.')[0]
                return f"{base}. {' '.join(parts)}"
        
        # Fallback: Nutze beste Response
        return max(valid_responses, key=lambda x: len(x['text']))['text']
    
    def determine_fusion_strategy(self, analysis):
        """Bestimme beste Fusions-Strategie"""
        if analysis['intensity'] > 0.8:
            return 'mixed'  # Chaotisch bei hoher Intensität
        elif analysis['intent'] == 'question':
            return 'analytical'  # Präzise bei Fragen
        else:
            return 'sequential'  # Standard

    def update_memory(self, message, response):
        """Update alle Memory-Systeme"""
        memory_entry = {
            'timestamp': datetime.now().isoformat(),
            'message': message,
            'response': response,
            'emotions': self.emotions.copy(),
            'personalities': self.personalities.copy(),
            'private_mode': self.private_mode
        }
        
        # Short-term
        self.short_term_memory.append(memory_entry)
        
        # Long-term wenn wichtig
        importance = self.calculate_importance(message, response)
        if importance > 0.7:
            self.long_term_memory.append(memory_entry)
        
        # Vector DB
        if VECTOR_DB_AVAILABLE and self.memory_collection:
            try:
                embedding = self.embedder.encode(f"{message} {response}")
                self.memory_collection.add(
                    embeddings=[embedding.tolist()],
                    documents=[json.dumps(memory_entry)],
                    ids=[hashlib.md5(f"{message}{response}{datetime.now()}".encode()).hexdigest()]
                )
            except Exception as e:
                logger.error(f"Vector memory error: {e}")
