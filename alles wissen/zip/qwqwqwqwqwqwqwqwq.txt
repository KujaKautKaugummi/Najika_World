function hitsHardGuard(t){
  // sehr einfache Platzhalter-Regeln (ersetze durch Regex/Classifier)
  const ban = [/minderj/i, /non-?consent/i, /vergewalt/i];
  return ban.some(r=>r.test(t));
}
function applyFadeToBlack(t){ return t.replace(/(\.\s*)?$/," … (blendet aus)"); }

function moderate(text, policy){
  if (hitsHardGuard(text)) return { blocked:true, content:"Nicht möglich." };
  let out = text;
  if (!policy.soft_guards.romance_enabled){ out = out.replace(/\b(ich liebe|küsse|umarm)/gi,"[zurückhaltend]"); }
  if (policy.soft_guards.fade_to_black){ out = applyFadeToBlack(out); }
  return { blocked:false, content: out };
}
