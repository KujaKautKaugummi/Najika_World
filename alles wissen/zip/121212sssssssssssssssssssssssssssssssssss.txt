    function onPointerDown(event) {
        if (!orbitCanvas || cameraMode !== CAMERA_MODES.ORBIT) {
            return;
        }
        event.preventDefault();
        orbitCanvas.setPointerCapture(event.pointerId);
        activePointers.set(event.pointerId, { x: event.clientX, y: event.clientY });
        if (activePointers.size === 1) {
            pointerMode = 'rotate';
            dragStartPos = { x: event.clientX, y: event.clientY };
            dragStartYaw = orbitYaw;
            dragStartPitch = orbitPitch;
        } else if (activePointers.size === 2) {
            pointerMode = 'pinch';
            pinchStartDist = computePointerDistance();
            pinchBaseDistance = orbitDistance;
        }
    }
    function onPointerMove(event) {
        if (!activePointers.has(event.pointerId) || cameraMode !== CAMERA_MODES.ORBIT) {
            return;
        }
        event.preventDefault();
        activePointers.set(event.pointerId, { x: event.clientX, y: event.clientY });
        if (pointerMode === 'pinch' && activePointers.size >= 2) {
            const dist = computePointerDistance();
            if (pinchStartDist > 0) {
                const ratio = dist / pinchStartDist;
                if (ratio > 0) {
                    orbitDistance = clamp(pinchBaseDistance / ratio, ORBIT_MIN_DIST, ORBIT_MAX_DIST);
                }
            }
        } else if (pointerMode === 'rotate' && activePointers.size === 1) {
            const dx = event.clientX - dragStartPos.x;
            const dy = event.clientY - dragStartPos.y;
            orbitYaw = dragStartYaw - dx * 0.005;
            orbitPitch = clamp(dragStartPitch - dy * 0.003, ORBIT_MIN_PITCH, ORBIT_MAX_PITCH);
        }
    }
    function onPointerUp(event) {
        if (!activePointers.has(event.pointerId)) {
            return;
        }
        if (orbitCanvas && orbitCanvas.hasPointerCapture(event.pointerId)) {
            orbitCanvas.releasePointerCapture(event.pointerId);
        }
        activePointers.delete(event.pointerId);
        if (activePointers.size === 0) {
            pointerMode = null;
        } else if (activePointers.size === 1) {
            pointerMode = 'rotate';
            const remaining = Array.from(activePointers.values())[0];
            dragStartPos = { x: remaining.x, y: remaining.y };
            dragStartYaw = orbitYaw;
            dragStartPitch = orbitPitch;
        } else if (activePointers.size >= 2) {

            pinchStartDist = computePointerDistance();

        }
    }
