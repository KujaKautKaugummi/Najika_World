    function createInstance(model, options = {}) {

            clone.scale.multiplyScalar(factor);

        if (options.rotation) {
            const rot = normalizeVector(options.rotation, 0);
            clone.rotation.set(rot[0], rot[1], rot[2]);
        }
        const bbox = new THREE.Box3().setFromObject(clone);

        if (options.position) {
            const pos = normalizeVector(options.position, 0);
            clone.position.x += pos[0];
            clone.position.y += pos[1];
            clone.position.z += pos[2];

        const eyeTarget = characterGroup.position.clone().add(new THREE.Vector3(0, characterHeight * 0.85, 0));

            const eyePosition = eyeTarget.clone().add(forwardDir.clone().multiplyScalar(1.8));

    async function toggleWebcam(videoEl) {
        if (webcamStream) {
            webcamStream.getTracks().forEach(track => track.stop());
            webcamStream = null;
            if (webcamElement) {
                webcamElement.srcObject = null;
            }
            if (videoEl) {
                videoEl.srcObject = null;
            }
            return false;
        }
        if (!navigator.mediaDevices || !navigator.mediaDevices.getUserMedia) {
            alert('Webcam wird nicht unterstützt.');
            return false;
        }
        try {
            const stream = await navigator.mediaDevices.getUserMedia({ video: true, audio: false });
            webcamStream = stream;
            webcamElement = videoEl || webcamElement;
            if (webcamElement) {
                webcamElement.srcObject = stream;
            }
            return true;
        } catch (error) {
            console.warn('Webcam Zugriff fehlgeschlagen', error);
            alert('Webcam konnte nicht gestartet werden.');
            return false;
        }
    }

    function normalizePalette(palette) {
        const result = {};
        if (palette.background !== undefined) {
            result.background = parseColor(palette.background, DEFAULT_PALETTE.background);
        }
        if (palette.fog !== undefined) {
            result.fog = parseColor(palette.fog, DEFAULT_PALETTE.fog);
        }
        if (palette.floor !== undefined) {
            result.floor = parseColor(palette.floor, DEFAULT_PALETTE.floor);
        }
        if (palette.primary !== undefined) {
            result.primary = parseColor(palette.primary, DEFAULT_PALETTE.primary);
        }
        if (palette.secondary !== undefined) {
            result.secondary = parseColor(palette.secondary, DEFAULT_PALETTE.secondary);
        }
        if (palette.dir !== undefined) {
            result.dir = parseColor(palette.dir, DEFAULT_PALETTE.dir);
        }
        if (palette.ambient !== undefined) {
            result.ambient = palette.ambient;
        }
        if (palette.dirIntensity !== undefined) {
            result.dirIntensity = palette.dirIntensity;
        }
        return result;
    }

    function parseColor(value, fallback) {
        if (value === null || value === undefined) {
            return fallback;
        }
        if (typeof value === 'number') {
            return value;
        }
        if (typeof value === 'string') {
            const trimmed = value.trim();
            if (trimmed.startsWith('#')) {
                return parseInt(trimmed.slice(1), 16);
            }
            if (trimmed.startsWith('0x') || trimmed.startsWith('0X')) {
                return parseInt(trimmed.slice(2), 16);
            }
            const numeric = parseInt(trimmed, 16);
            if (!Number.isNaN(numeric)) {
                return numeric;
            }
        }
        return fallback;
    }
    function normalizeVector(arr, fallback) {
        if (!Array.isArray(arr)) {
            return [fallback, fallback, fallback];
        }
        return [
            Number(arr[0] ?? fallback),
            Number(arr[1] ?? fallback),
            Number(arr[2] ?? fallback)
        ];
    }

            { rotation: [0, 0, 0], axis: 'front' },
            { rotation: [0, Math.PI, 0], axis: 'back' },
            { rotation: [0, Math.PI / 2, 0], axis: 'left' },
            { rotation: [0, -Math.PI / 2, 0], axis: 'right' }
        ];
        let success = false;
        placements.forEach(placement => {
            const result = createInstance(model, {
                targetSpan: span,

            });

            } else if (placement.axis === 'left') {
                wall.position.x -= (span / 2) + (size.x / 2);
            } else if (placement.axis === 'right') {
                wall.position.x += (span / 2) + (size.x / 2);

    function createInstance(model, options = {}) {
        const clone = model.clone(true);
        clone.traverse(node => {
            if (node.isMesh) {
                if (node.material) {
                    node.material = node.material.clone();
                    node.material.side = THREE.DoubleSide;
                    if (node.material.map) {
                        node.material.map.encoding = THREE.sRGBEncoding;
                        node.material.map.needsUpdate = true;
                    }
                }
                node.castShadow = true;
                node.receiveShadow = true;
            }
        });
        if (options.scale) {
            if (Array.isArray(options.scale) && options.scale.length === 3) {
                clone.scale.set(options.scale[0], options.scale[1], options.scale[2]);
            } else if (typeof options.scale === 'number') {
                clone.scale.multiplyScalar(options.scale);
            }
        }
        if (options.targetSpan) {
            const sizeBefore = new THREE.Box3().setFromObject(clone).getSize(new THREE.Vector3());
            const spanBefore = Math.max(sizeBefore.x, sizeBefore.z, 0.0001);
            const factor = options.targetSpan / spanBefore;
            clone.scale.multiplyScalar(factor);
        }
        if (options.rotation) {
            const rot = normalizeVector(options.rotation, 0);
            clone.rotation.set(rot[0], rot[1], rot[2]);
        }
        const bbox = new THREE.Box3().setFromObject(clone);
        const center = bbox.getCenter(new THREE.Vector3());
        const size = bbox.getSize(new THREE.Vector3());
        clone.position.sub(center);
        clone.position.y += size.y / 2;
        if (options.position) {
            const pos = normalizeVector(options.position, 0);
            clone.position.x += pos[0];
            clone.position.y += pos[1];
            clone.position.z += pos[2];
        }
        if (options.tint) {
            applyTint(clone, parseColor(options.tint, null));
        }
        return { object: clone, size };
    }
    function applyTint(object, color) {
        if (color === null || color === undefined) {
            return;
        }
        object.traverse(node => {
            if (node.isMesh && node.material && node.material.color) {
                node.material.color.setHex(color);
                if (node.material.emissive) {
                    node.material.emissive.setHex(color);
                    node.material.emissiveIntensity = Math.min(node.material.emissiveIntensity || 0.4, 0.6);
                }
            }
        });
    }

    function getMovementVector() {
        let forward = 0;
        let strafe = 0;
        if (activeKeys.has('KeyW') || activeKeys.has('ArrowUp')) forward += 1;
        if (activeKeys.has('KeyS') || activeKeys.has('ArrowDown')) forward -= 1;
        if (activeKeys.has('KeyA') || activeKeys.has('ArrowLeft')) strafe -= 1;
        if (activeKeys.has('KeyD') || activeKeys.has('ArrowRight')) strafe += 1;
        const vector = new THREE.Vector3(strafe, 0, -forward);
        if (vector.lengthSq() > 0) {
            vector.normalize();
        }
        return vector;
    }
    function updateCharacter(delta) {
        if (!characterGroup) {
            return;
        }
        const move = getMovementVector();
        if (move.lengthSq() > 0) {
            const motion = move.clone().multiplyScalar(characterSpeed * delta);
            characterGroup.position.add(motion);
            characterGroup.position.y = 0;
            characterHeading = Math.atan2(move.x, -move.z);
            characterGroup.rotation.y = characterHeading;
        }
    }
    function updateCamera(delta) {
        if (!camera) {
            return;
        }
        if (cameraMode === CAMERA_MODES.ORBIT || !characterGroup) {
            orbitTime += delta;
            const t = orbitTime * 0.5;
            camera.position.x = Math.sin(t * 0.8) * 8;
            camera.position.z = 12 + Math.cos(t * 0.6) * 3;
            camera.position.y = 5 + Math.sin(t) * 0.8;
            camera.lookAt(0, 2, 0);
            return;
        }
        const forwardDir = new THREE.Vector3(Math.sin(characterHeading), 0, -Math.cos(characterHeading));
        const eyeTarget = characterGroup.position.clone().add(new THREE.Vector3(0, characterHeight * 0.85, 0));
        if (cameraMode === CAMERA_MODES.THIRD) {
            const followOffset = forwardDir.clone().multiplyScalar(-18);
            const desiredPosition = eyeTarget.clone().add(new THREE.Vector3(0, 4, 0)).add(followOffset);
            camera.position.lerp(desiredPosition, Math.min(delta * 6, 1));
