        self.long_term = []   # Persistent Storage
        self.emotional_tags = {}  # Gefühls-Kontext
        
    def store_interaction(self, user_input, najika_response, emotion_level):
        memory_entry = {
            "timestamp": datetime.now(),
            "user": user_input,
            "najika": najika_response,
            "emotion": emotion_level,
            "context_tags": self.extract_context(user_input)
        }
        self.long_term.append(memory_entry)
        
    def recall_similar(self, current_input):
        # Najika erinnert sich an ähnliche Gespräche
        similar = []
        for memory in self.long_term:
            if self.similarity_score(current_input, memory["user"]) > 0.7:
                similar.append(memory)
        return similar

# najika_privacy_detector.py
import cv2
import psutil
import pyaudio
import numpy as np
import threading
import time

class PrivacyDetector:
    def __init__(self):
        self.is_alone = False
        self.camera_active = False
        self.mic_active = False
        self.monitoring = False
        
    def start_monitoring(self):
        self.monitoring = True
        threading.Thread(target=self.check_camera, daemon=True).start()
        threading.Thread(target=self.check_microphone, daemon=True).start()
        threading.Thread(target=self.check_apps, daemon=True).start()
        
    def check_camera(self):
        # Gesichtserkennung
        face_cascade = cv2.CascadeClassifier(cv2.data.haarcascades + 'haarcascade_frontalface_default.xml')
        cap = cv2.VideoCapture(0)
        
        while self.monitoring:
            try:
                ret, frame = cap.read()
                if ret:
                    gray = cv2.cvtColor(frame, cv2.COLOR_BGR2GRAY)
                    faces = face_cascade.detectMultiScale(gray, 1.3, 5)
                    
                    # Mehr als 1 Gesicht = nicht alleine
                    self.is_alone = len(faces) <= 1
                    
                time.sleep(2)  # Check alle 2 Sekunden
            except:
                pass
                
        cap.release()
        
    def check_microphone(self):
        # Umgebungsgeräusche analysieren
        p = pyaudio.PyAudio()
        
        try:
            stream = p.open(format=pyaudio.paInt16, channels=1, 
                           rate=44100, input=True, frames_per_buffer=1024)
            
            while self.monitoring:
                data = np.frombuffer(stream.read(1024), dtype=np.int16)
                volume = np.abs(data).mean()
                
                # Zu viel Lärm = nicht alleine
                if volume > 1000:  # Schwellwert anpassbar
                    self.is_alone = False
                    
                time.sleep(1)
                
        except:
            pass
        finally:
            if 'stream' in locals():
                stream.close()
            p.terminate()

# najika_scanner.py - KOMPLETTER ORDNER-SCANNER
import os
import json
from datetime import datetime

def scan_najika_directory():
    base_path = "C:\\NajikaAI"
    
    print("🔍 SCANNE NAJIKA AI ORDNER...")
    print("=" * 50)
    
    if not os.path.exists(base_path):
        print("❌ C:\\NajikaAI existiert nicht!")
        return
    
    # Vollständige Struktur erfassen
    structure = {}
    
    for root, dirs, files in os.walk(base_path):
        # Relativen Pfad berechnen
        rel_path = os.path.relpath(root, base_path)
        if rel_path == ".":
            rel_path = "ROOT"
        
        structure[rel_path] = {
            "ordner": dirs,
            "dateien": files,
            "absoluter_pfad": root
        }
        
        print(f"\n📁 {rel_path}:")
        print(f"   Unterordner: {dirs}")
        print(f"   Dateien: {files}")
        
        # Dateien analysieren
        for file in files:
            file_path = os.path.join(root, file)
            try:
                size = os.path.getsize(file_path)
                modified = datetime.fromtimestamp(os.path.getmtime(file_path))
                print(f"   📄 {file} ({size} bytes, {modified.strftime('%Y-%m-%d %H:%M')})")
            except:
                print(f"   📄 {file} (Fehler beim Lesen)")
    
    # JSON-Report erstellen
    with open("najika_structure.json", "w", encoding="utf-8") as f:
        json.dump(structure, f, indent=2, ensure_ascii=False)
    
    print(f"\n✅ Struktur gespeichert in: najika_structure.json")
    return structure

if __name__ == "__main__":
    scan_najika_directory()

    <script src="https://cdnjs.cloudflare.com/ajax/libs/socket.io/4.0.1/socket.io.js"></script>
    <script>
        const socket = io();
        
        function addMessage(sender, message, isUser = false) {
            const chatArea = document.getElementById('chatArea');
            const messageDiv = document.createElement('div');
            messageDiv.className = `message ${isUser ? 'user-message' : 'najika-message'}`;
            messageDiv.innerHTML = `<strong>${sender}:</strong> ${message}`;
            chatArea.appendChild(messageDiv);
            chatArea.scrollTop = chatArea.scrollHeight;
        }
        
        function sendMessage() {
            const input = document.getElementById('messageInput');
            const message = input.value.trim();
            if (message) {
                addMessage('Du', message, true);
                socket.emit('chat', {message: message});
                input.value = '';
            }
        }
        
        function handleEnter(event) {
            if (event.key === 'Enter') {
                sendMessage();
            }
        }
        
        function careAction(action) {
            socket.emit('care_action', {action: action});
        }
        
        socket.on('najika_says', function(data) {
            addMessage('Najika', data.message);
        });
        
        socket.on('najika_reaction', function(data) {
            addMessage('Najika', data.message);
            // Status aktualisieren
            if (data.status) {
                document.getElementById('energy-status').textContent = data.status.energy + '%';
                if (data.bond_level) {
                    document.getElementById('bond-status').textContent = data.bond_level + '/100';
                }
            }
        });
        
        socket.on('connect', function() {
            console.log('🔗 Najika Digivice verbunden!');
        });
    </script>
